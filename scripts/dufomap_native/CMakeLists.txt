cmake_minimum_required(VERSION 3.17)
project(lio_sam_dufomap_native LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 CONFIG REQUIRED)
find_package(TBB REQUIRED)
find_package(OpenMP REQUIRED)
find_library(LZ4_LIBRARY NAMES lz4 REQUIRED)
find_path(LZF_INCLUDE_DIR NAMES liblzf/lzf.h REQUIRED)
find_library(LZF_LIBRARY NAMES lzf REQUIRED)
if(NOT EXISTS "${UFOMAP_SOURCE}/include/ufo/map/ufomap.hpp")
  message(FATAL_ERROR "UFOMAP_SOURCE must name the pinned UFOMap source checkout")
endif()
pybind11_add_module(dufomap NO_EXTRAS dufomap.cpp)
target_compile_features(dufomap PRIVATE cxx_std_20)
target_compile_definitions(dufomap PRIVATE UFO_PARALLEL=1)
target_include_directories(dufomap PRIVATE "${UFOMAP_SOURCE}/include" "${LZF_INCLUDE_DIR}")
target_link_libraries(dufomap PRIVATE TBB::tbb OpenMP::OpenMP_CXX "${LZ4_LIBRARY}" "${LZF_LIBRARY}")
# Upstream unconditionally includes an x86 header, but every intrinsic is gated
# on __BMI2__. An empty include shim selects upstream's portable bit algorithms.
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  target_include_directories(dufomap BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/arm64")
endif()
install(TARGETS dufomap LIBRARY DESTINATION .)
